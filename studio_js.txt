/* ------------------------------------------------------------------ single-page trip studio */
function routeIds() {
  const ids = (state.routeOrder || []).filter((id) => state.selected.has(id));
  if (ids.length !== state.selected.size) {
    const extra = [...state.selected.keys()].filter((id) => !ids.includes(id));
    state.routeOrder = [...ids, ...extra];
    return state.routeOrder.filter((id) => state.selected.has(id));
  }
  return ids;
}

function autoEnd() {
  const ids = routeIds();
  const days = ids.length
    ? ids.reduce((n, id) => n + ((destById(id) || {}).recommendedDays || 2), 0)
    : 3;
  const d = new Date(state.startDate + 'T12:00:00');
  d.setDate(d.getDate() + Math.max(1, Math.min(21, days)) - 1);
  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
}

function totalDays() {
  if (!state.endDate) return 3;
  const start = new Date(state.startDate + 'T12:00:00');
  const end = new Date(state.endDate + 'T12:00:00');
  const diff = Math.round((end - start) / 86400000) + 1;
  return Math.max(1, Math.min(21, diff || 1));
}

function allocateDays(total, ids) {
  if (!ids.length) return [];
  const target = Math.max(ids.length, Math.min(21, total));
  const rec = ids.map((id) => (destById(id) || {}).recommendedDays || 2);
  let remaining = target - ids.length;
  const sum = rec.reduce((a, b) => a + b, 0);
  const raw = rec.map((r) => (remaining * r) / sum);
  const alloc = raw.map((v) => Math.floor(v));
  let left = remaining - alloc.reduce((a, b) => a + b, 0);
  const order = raw.map((v, i) => ({ i, frac: v - Math.floor(v) })).sort((a, b) => b.frac - a.frac);
  for (let k = 0; k < left; k++) alloc[order[k % order.length].i]++;
  return alloc.map((n) => n + 1);
}

function autoTitle(ids) {
  const names = ids.map((id) => (destById(id) || {}).name).filter(Boolean);
  if (!names.length) return 'Your Myanmar trip';
  if (names.length === 1) return names[0] + ' Escape';
  if (names.length === 2) return names[0] + ' & ' + names[1];
  return names[0] + ' & ' + (names.length - 1) + ' more stops';
}

function renderStudioDestGrid() {
  const host = $('#studioDestGrid');
  if (!host) return;
  const q = (state.plannerQuery || '').toLowerCase().trim();
  const list = state.destinations.filter((d) => {
    if (!q) return true;
    return [d.name, d.nameMm, d.region, d.tagline, ...(d.places || []).map((p) => p.name)].join(' ').toLowerCase().includes(q);
  });
  host.innerHTML = list.length ? list.map((d) => {
    const added = state.selected.has(d.id);
    return `<article class="studio-dest ${added ? 'is-added' : ''}">
      <div class="studio-dest-media">${sceneFor(d.id)}</div>
      <div class="studio-dest-body">
        <div class="studio-dest-region">${esc(d.region)}</div>
        <h4>${esc(d.name)}</h4>
        <p>${esc(d.tagline)}</p>
        <div class="studio-dest-foot"><span>${d.recommendedDays} days suggested</span><button data-studio-add="${esc(d.id)}">${added ? 'Remove' : '+ Add'}</button></div>
      </div>
    </article>`;
  }).join('') : '<div class="studio-empty">No destinations match that search.</div>';
}

function bindStudio() {
  const host = $('#plannerStudio');
  if (!host) return;
  const title = $('#studioTitle');
  if (title) title.addEventListener('input', () => {
    state.tripTitle = title.value.trim();
    const t = $('#studioSummaryTitle');
    if (t) t.textContent = state.tripTitle || autoTitle(routeIds());
  });
  const budget = $('#studioBudget');
  if (budget) budget.addEventListener('input', () => {
    const raw = budget.value.replace(/[^0-9]/g, '').slice(0, 9);
    budget.value = raw;
    state.budget = raw ? Number(raw) : 0;
    const b = $('#studioSummaryBudget');
    if (b) b.textContent = money(state.budget);
  });
  const search = $('#studioSearch');
  if (search) search.addEventListener('input', () => {
    state.plannerQuery = search.value;
    renderStudioDestGrid();
  });
  const start = $('#studioStart');
  if (start) start.addEventListener('change', () => {
    state.startDate = start.value || state.startDate;
    if (!state.endTouched || state.endDate < state.startDate) state.endDate = autoEnd();
    renderPlannerStudio();
    syncQuickFields();
  });
  const end = $('#studioEnd');
  if (end) end.addEventListener('change', () => {
    state.endTouched = true;
    state.endDate = end.value || autoEnd();
    if (state.endDate < state.startDate) state.endDate = state.startDate;
    renderPlannerStudio();
  });
  const currency = $('#studioCurrency');
  if (currency) currency.addEventListener('change', () => {
    state.currency = currency.value;
    renderPlannerStudio();
    if (state.itinerary) renderItinerary(state.itinerary);
  });
  const gen = $('#studioGenerate');
  if (gen) gen.addEventListener('click', studioGenerate);

  $$('[data-studio-count]', host).forEach((b) => b.addEventListener('click', () => {
    const which = b.dataset.studioCount;
    const delta = Number(b.dataset.delta);
    if (which === 'adults') state.adults = Math.max(1, Math.min(12, state.adults + delta));
    if (which === 'children') state.children = Math.max(0, Math.min(10, state.children + delta));
    renderPlannerStudio();
    syncQuickFields();
  }));
  $$('[data-studio-style]', host).forEach((b) => b.addEventListener('click', () => { state.style = b.dataset.studioStyle; renderPlannerStudio(); }));
  $$('[data-studio-transport]', host).forEach((b) => b.addEventListener('click', () => { state.transport = b.dataset.studioTransport; renderPlannerStudio(); }));
  $$('[data-studio-pace]', host).forEach((b) => b.addEventListener('click', () => { state.pace = b.dataset.studioPace; renderPlannerStudio(); }));
  $$('[data-studio-interest]', host).forEach((b) => b.addEventListener('click', () => {
    const id = b.dataset.studioInterest;
    if (state.interests.has(id)) state.interests.delete(id); else state.interests.add(id);
    renderPlannerStudio();
  }));
  $$('[data-studio-add]', host).forEach((b) => b.addEventListener('click', () => {
    toggleDestination(b.dataset.studioAdd);
  }));
  $$('[data-remove-stop]', host).forEach((b) => b.addEventListener('click', () => {
    toggleDestination(b.dataset.removeStop);
  }));
  $$('[data-move]', host).forEach((b) => b.addEventListener('click', () => {
    const stop = b.closest('[data-stop]');
    const id = stop && stop.dataset.stop;
    const delta = Number(b.dataset.move);
    const ids = routeIds();
    const i = ids.indexOf(id);
    const j = i + delta;
    if (i < 0 || j < 0 || j >= ids.length) return;
    const copy = [...ids];
    [copy[i], copy[j]] = [copy[j], copy[i]];
    state.routeOrder = copy;
    renderPlannerStudio();
  }));

  $$('.studio-stop', host).forEach((stop) => {
    stop.addEventListener('dragstart', (e) => {
      e.dataTransfer.setData('text/plain', stop.dataset.stop);
      e.dataTransfer.effectAllowed = 'move';
      stop.classList.add('dragging');
    });
    stop.addEventListener('dragend', () => stop.classList.remove('dragging'));
    stop.addEventListener('dragover', (e) => { e.preventDefault(); e.dataTransfer.dropEffect = 'move'; });
    stop.addEventListener('drop', (e) => {
      e.preventDefault();
      const dragged = e.dataTransfer.getData('text/plain');
      const target = stop.dataset.stop;
      if (!dragged || dragged === target) return;
      const ids = routeIds().filter((id) => id !== dragged);
      const at = ids.indexOf(target);
      ids.splice(at < 0 ? ids.length : at, 0, dragged);
      state.routeOrder = ids;
      renderPlannerStudio();
    });
  });
}

function studioDistance(ids) {
  const points = ids.map(destById).filter(Boolean).map((d) => [d.lat, d.lon]);
  let km = 0;
  for (let i = 1; i < points.length; i++) {
    const [lat1, lon1] = points[i - 1];
    const [lat2, lon2] = points[i];
    const R = 6371;
    const dLat = (lat2 - lat1) * Math.PI / 180;
    const dLon = (lon2 - lon1) * Math.PI / 180;
    const a = Math.sin(dLat / 2) ** 2 + Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) * Math.sin(dLon / 2) ** 2;
    km += 2 * R * Math.asin(Math.sqrt(a));
  }
  const speeds = { private_car: 55, bus: 45, flight: 520, boat: 25, mixed: 65 };
  const speed = speeds[state.transport] || 55;
  const hours = km / speed + (state.transport === 'flight' ? points.length * 1.5 : points.length * 0.25);
  return { km: Math.round(km), hours: Math.max(0, Math.round(hours * 10) / 10) };
}

function renderStudioMap() {
  const el = document.getElementById('studioMap');
  if (!el) return;
  const ids = routeIds();
  const meta = document.getElementById('studioRouteTotal');
  const mapMeta = document.getElementById('studioMapMeta');
  if (!ids.length) {
    el.innerHTML = '<div class="studio-empty">Add destinations to see the route map.</div>';
    if (mapMeta) mapMeta.textContent = 'No route yet';
    if (meta) meta.textContent = '';
    return;
  }
  const dist = studioDistance(ids);
  if (meta) meta.innerHTML = `<span>Route estimate</span><b>≈ ${dist.km} km · ${dist.hours} h by ${esc(transportName(state.transport))}</b>`;
  if (mapMeta) mapMeta.textContent = `≈ ${dist.km} km · ${dist.hours} h`;
  if (!window.L) {
    el.innerHTML = routeMapSvg({ destinationIds: ids }, true);
    return;
  }
  if (state.studioMap) { try { state.studioMap.remove(); } catch { /* ignore */ } state.studioMap = null; }
  el.innerHTML = '';
  const map = L.map(el, { zoomControl: true, scrollWheelZoom: false });
  L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', { maxZoom: 18, attribution: '&copy; OpenStreetMap contributors' }).addTo(map);
  const route = ids.map(destById).filter(Boolean);
  const pts = route.map((d) => [d.lat, d.lon]);
  if (pts.length > 1) L.polyline(pts, { color: '#d9a53a', weight: 4, dashArray: '7 7' }).addTo(map);
  route.forEach((d, i) => {
    L.marker([d.lat, d.lon], {
      icon: L.divIcon({ className: 'rmap-marker ' + (i === 0 ? 'start' : ''), html: '<span>' + (i + 1) + '</span><b>' + esc(d.name) + '</b>', iconSize: [0, 0] }),
    }).addTo(map).bindPopup('<b>' + esc(d.name) + '</b><br>' + esc(d.region));
  });
  if (pts.length > 1) map.fitBounds(pts, { padding: [40, 40] });
  else map.setView(pts[0], 8);
  state.studioMap = map;
  setTimeout(() => map.invalidateSize(), 80);
}

function studioGenerate() {
  const ids = routeIds();
  if (!ids.length) { toast('Choose at least one destination.', true); return; }
  if (totalDays() > 21) { toast('Trips are limited to 21 days. Shorten the end date.', true); return; }
  if (state.endDate < state.startDate) { toast('The end date must be after the start date.', true); return; }
  state.tripTitle = $('#studioTitle') ? $('#studioTitle').value.trim() : state.tripTitle;
  generateItinerary({ fromStudio: true });
}
